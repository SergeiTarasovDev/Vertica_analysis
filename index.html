    #!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
Vertica bootstrap — восстановление схем, ролей, привилегий, ресурсных пулов и пользователей.

Скрипт создаёт объекты из встроенных списков, собранных из DDL-скриптов DBeaver
(каталог General\Scripts). Пароли пользователей генерируются автоматически и
складываются в отдельный файл кредов.

Установка зависимости:
    pip install vertica-python

Запуск (примеры):
    # 1) холостой прогон: в БД ничего не делает, печатает SQL и пишет файл кредов
    python vertica_bootstrap.py --dry-run --creds-out vertica_creds.csv

    # 2) боевой прогон
    python vertica_bootstrap.py --host 10.101.1.31 --port 5433 --user dbadmin --password "***" --creds-out vertica_creds.csv

Флаги:
    --reset-passwords   назначать новый пароль и уже существующим пользователям (ALTER USER)
    --no-pools          не создавать ресурсные пулы
    --no-users          не создавать пользователей
    --schemas-only      только схемы и роли
    --dry-run           не подключаться к БД, показать SQL и сформировать креды

Логика:
    * Скрипт ИДЕМПОТЕНТЕН и НЕ выполняет REVOKE/DROP — безопасен для повторного запуска.
    * CREATE SCHEMA / ROLE — с IF NOT EXISTS; CREATE USER / RESOURCE POOL — по try/except.
    * Пользователю, который уже существует, пароль по умолчанию НЕ меняется
      (в отчёте он помечается как "exists"), если не передан --reset-passwords.
    * Исключены явно тестовые сущности (test_*, analyst1/user1, my_schema, scr_, read_only_role и т.п.).
"""

def q(name: str) -> str:
    """Кавычит идентификатор Vertica."""
    return '"' + str(name).replace('"', '""') + '"'


def literal(value: str) -> str:
    """Кавычит строковый литерал."""
    return "'" + str(value).replace("'", "''") + "'"


def main() -> int:
    ap = argparse.ArgumentParser(description="Vertica bootstrap (schemas/roles/pools/users)")
    ap.add_argument("--host", default="127.0.0.1", help="хост Vertica")
    ap.add_argument("--port", type=int, default=5433, help="порт Vertica")
    ap.add_argument("--user", default="dbadmin", help="административный пользователь")
    ap.add_argument("--password", default="", help="пароль администратора")
    ap.add_argument("--timeout", type=int, default=60, help="таймаут подключения, сек")
    ap.add_argument("--creds-out", default="vertica_creds.csv", help="файл со сгенерированными паролями")
    ap.add_argument("--reset-passwords", action="store_true", help="также сбросить пароли существующих пользователей")
    ap.add_argument("--no-pools", action="store_true", help="не создавать ресурсные пулы")
    ap.add_argument("--no-users", action="store_true", help="не создавать пользователей")
    ap.add_argument("--schemas-only", action="store_true", help="только схемы и роли")
    ap.add_argument("--dry-run", action="store_true", help="не подключаться к БД")
    args = ap.parse_args()

    conn = None
    cur = None
    if not args.dry_run:
        try:
            import vertica_python  # noqa: WPS433
        except ImportError:
            print("Не установлена зависимость. Выполните: pip install vertica-python", file=sys.stderr)
            return 2
        conn = vertica_python.connect(
            host=args.host,
            port=args.port,
            user=args.user,
            password=args.password,
            connection_timeout=args.timeout,
        )
        cur = conn.cursor()

    def run(sql: str, ignore_errors: bool = True) -> bool:
        """Выполняет SQL. В dry-run только печатает. Возвращает True при успехе."""
        if args.dry_run:
            print("SQL>", sql)
            return True
        try:
            cur.execute(sql)
            return True
        except Exception as exc:  # noqa: BLE001
            msg = str(exc).strip().splitlines()
            print(f"[warn] {msg[0] if msg else exc}", file=sys.stderr)
            print(f"       SQL: {sql}", file=sys.stderr)
            if not ignore_errors:
                raise
            return False

    # ---------------- 1. Схемы ----------------
    print(f"== Схемы: {len(SCHEMAS)}")
    for s in SCHEMAS:
        run(f"CREATE SCHEMA IF NOT EXISTS {q(s)};")

    print(f"== Наследование привилегий схемам: {len(SCHEMA_INHERIT)}")
    for s in SCHEMA_INHERIT:
        run(f"ALTER SCHEMA {q(s)} DEFAULT INCLUDE SCHEMA PRIVILEGES;")

    # ---------------- 2. Роли и привилегии ----------------
    print(f"== Роли: {len(ROLES)}")
    for role, grants in sorted(ROLES.items()):
        run(f"CREATE ROLE IF NOT EXISTS {q(role)};")
        for schema, priv in grants:
            run(f"GRANT {priv} ON SCHEMA {q(schema)} TO {q(role)};")

    # ---------------- 3. Ресурсные пулы ----------------
    if not args.schemas_only and not args.no_pools:
        print(f"== Ресурсные пулы: {len(POOLS)}")
        for pool, ddl in sorted(POOLS.items()):
            if not ddl.rstrip().endswith(";"):
                ddl = ddl.rstrip() + ";"
            run(ddl)

    # ---------------- 4. Пользователи ----------------
    creds = []
    if not args.schemas_only and not args.no_users:
        print(f"== Пользователи: {len(USERS)}")
        for user, info in sorted(USERS.items()):
            password = secrets.token_urlsafe(24)
            created = run(f"CREATE USER {q(user)} IDENTIFIED BY {literal(password)};")
            status = "created"
            if not created:
                if args.reset_passwords:
                    run(f"ALTER USER {q(user)} IDENTIFIED BY {literal(password)};")
                    status = "reset"
                else:
                    status = "exists"

            for role in info.get("roles", []):
                run(f"GRANT {q(role)} TO {q(user)};")

            default_role = info.get("default_role") or ""
            if default_role:
                run(f"ALTER USER {q(user)} DEFAULT ROLE {default_role};")

            for pool in reversed(info.get("pools", [])):
                run(f"GRANT USAGE ON RESOURCE POOL {q(pool)} TO {q(user)};")
                run(f"ALTER USER {q(user)} RESOURCE POOL {q(pool)};")

            if args.dry_run or status in ("created", "reset"):
                creds.append({
                    "user": user,
                    "password": password,
                    "status": status,
                    "pool": ",".join(info.get("pools", [])),
                    "roles": ",".join(info.get("roles", [])),
                })

        if args.creds_out:
            with open(args.creds_out, "w", newline="", encoding="utf-8") as fh:
                writer = csv.DictWriter(fh, fieldnames=["user", "password", "status", "pool", "roles"])
                writer.writeheader()
                writer.writerows(creds)
            print(f"== Креды сохранены: {args.creds_out} ({len(creds)} записей)")

    if conn is not None:
        conn.commit()
        conn.close()
        print("== Готово.")
    return 0
